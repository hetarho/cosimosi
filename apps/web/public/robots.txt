# The origin root belongs to the app, not to the blog served beneath it.
#
# The SPA answers every path with the same index.html, so without these Disallow lines a crawler would
# index the landing page's content under half a dozen wrong URLs. The list is enumerated rather than
# `Disallow: /` + allowlist, because the blog served at /blog/ must stay indexable — which also means an
# unknown path is not blocked, only unlisted: it still returns the shell, and the not-found screen is what
# a human sees. That is the same client-rendering limitation the deferred prerendering note records.
#
# Origin: see DEPLOY.md §1 (the deployment SSOT).

User-agent: *
Allow: /
Allow: /demo

# Authenticated product routes — the SPA serves the shell for all of them.
Disallow: /universe
Disallow: /diary
Disallow: /me
Disallow: /admin

# Auth surfaces: nothing here is a landing target, and a login form is not a search result.
Disallow: /login
Disallow: /signup
Disallow: /invite/

# Dev-only surfaces, reachable only behind the diagnostics flag.
Disallow: /test
Disallow: /design

Sitemap: https://cosimosi.haeram.me/sitemap.xml
Sitemap: https://cosimosi.haeram.me/blog/sitemap.xml
